// ---- mmvq.cu (kernel)
// engine2n: one-token MoE down projection (MUL_MAT_ID) followed by the weighted sum over the used experts. A block owns one
// output row: warp e computes expert slot e's row with the loop and warp sum mul_mat_vec_q uses on RDNA (one warp per row), and
// thread 0 sums the slots in moe_weighted_reduction_f32's order. Same values; the reduction kernel goes away.
template <ggml_type type>
static __global__ void __launch_bounds__(1024, 1) mul_mat_vec_q_moe_reduce(
        const void * vx, const void * vy, const int32_t * ids, const float * expert_scale, const float * weights, float * dst,
        const uint32_t ncols_x, const uint32_t nchannels_y, const uint32_t stride_row_x, const uint32_t stride_channel_x,
        const uint32_t stride_channel_y, const int n_expert_used) {
    constexpr int qk  = ggml_cuda_type_traits<type>::qk;
    constexpr int qi  = ggml_cuda_type_traits<type>::qi;
    constexpr int vdr = get_vdr_mmvq(type);
    constexpr int warp_size = 32;
    constexpr int blocks_per_iter = vdr * warp_size / qi;
    constexpr vec_dot_q_cuda_t vec_dot_q_cuda = get_vec_dot_q_cuda(type);
    __shared__ float vals[warp_size];

    const int lane = threadIdx.x % warp_size;
    const int e    = threadIdx.x / warp_size;
    const int row0 = blockIdx.x;
    const uint32_t channel_x = ids[e];
    const uint32_t channel_y = e % nchannels_y;
    const int blocks_per_row_x = ncols_x / qk;

    float tmp = 0.0f;
    const block_q8_1 * y = ((const block_q8_1 *) vy) + channel_y*stride_channel_y;
    const int kbx_offset = channel_x*stride_channel_x + row0*stride_row_x;
    for (int kbx = lane / (qi/vdr); kbx < blocks_per_row_x; kbx += blocks_per_iter) {
        const int kby = kbx * (qk/QK8_1);
        const int kqs = vdr * (lane % (qi/vdr));
        tmp += vec_dot_q_cuda(vx, &y[kby], kbx_offset + kbx, kqs);
    }
    tmp = warp_reduce_sum<warp_size>(tmp);
    if (lane == 0) {
        vals[e] = tmp;
    }
    __syncthreads();

    if (threadIdx.x == 0) {
        const float first_scale = expert_scale != nullptr ? expert_scale[0] : 1.0f;
        float       sum         = (vals[0] * first_scale) * weights[0];
        for (int expert = 1; expert < n_expert_used; ++expert) {
            const float scale = expert_scale != nullptr ? expert_scale[expert] : 1.0f;
            sum += (vals[expert] * scale) * weights[expert];
        }
        dst[row0] = sum;
    }
}

static void mul_mat_vec_q_moe_reduce_switch_type(
        const void * vx, const ggml_type type_x, const void * vy, const int32_t * ids, const float * expert_scale, const float * weights,
        float * dst, const int ncols_x, const int nrows_x, const int nchannels_y, const int stride_row_x, const int stride_channel_x,
        const int stride_channel_y, const int n_expert_used, cudaStream_t stream) {
#define MOE_REDUCE_CASE(t) case t: mul_mat_vec_q_moe_reduce<t><<<nrows_x, n_expert_used*32, 0, stream>>>(vx, vy, ids, expert_scale, \
        weights, dst, ncols_x, nchannels_y, stride_row_x, stride_channel_x, stride_channel_y, n_expert_used); break;
    switch (type_x) {
        MOE_REDUCE_CASE(GGML_TYPE_Q4_0)
        MOE_REDUCE_CASE(GGML_TYPE_Q5_0)
        MOE_REDUCE_CASE(GGML_TYPE_Q8_0)
        MOE_REDUCE_CASE(GGML_TYPE_IQ4_NL)
        MOE_REDUCE_CASE(GGML_TYPE_IQ4_XS)
        MOE_REDUCE_CASE(GGML_TYPE_Q4_K)
        MOE_REDUCE_CASE(GGML_TYPE_Q5_K)
        MOE_REDUCE_CASE(GGML_TYPE_Q6_K)
        default:
            GGML_ABORT("moe reduce: unsupported type");
    }
#undef MOE_REDUCE_CASE
}

// ---- ggml-cuda.cu (matcher)
// engine2n: one-token MoE down projection followed by the pattern moe_weighted_reduction fuses (MUL by the router weights,
// views, adds) in one kernel (mmvq.cu mul_mat_vec_q_moe_reduce, same values). Returns the nodes to skip, 0 if it does not
// match. GGML_CUDA_NO_MOE_DOWN_REDUCE=1 turns it off; GGML_CUDA_MOE_REDUCE_VERIFY=1 (debug) also runs the two ordinary kernels
// and compares the results bit for bit.
static bool ggml_cuda_tensors_overlap(const ggml_tensor * a, const ggml_tensor * b);
static int ggml_cuda_try_moe_down_reduce(ggml_backend_cuda_context & ctx, ggml_cgraph * cgraph, int i) {
    static const bool off = getenv("GGML_CUDA_NO_MOE_DOWN_REDUCE") != nullptr;
    ggml_tensor * down = cgraph->nodes[i];
    if (off || i + 1 >= cgraph->n_nodes) {
        return 0;
    }
    const ggml_tensor * w   = down->src[0];
    const ggml_tensor * y   = down->src[1];
    const ggml_tensor * ids = down->src[2];
    const int cc = ggml_cuda_info().devices[ctx.device].cc;
    if (!ggml_cuda_mmvq_warp_row_supported(w->type, cc) || down->type != GGML_TYPE_F32 || y->type != GGML_TYPE_F32 ||
            ids->type != GGML_TYPE_I32 || down->ne[2] != 1 || down->ne[3] != 1 || w->ne[3] != 1 || down->ne[1] > 32 ||
            !ggml_is_contiguous(down) || !ggml_node_has_n_uses(cgraph, i, 1)) {
        return 0;
    }
    ggml_cuda_moe_weighted_reduction_match match;
    if (!ggml_cuda_match_moe_weighted_reduction(cgraph, i + 1, match) || match.experts != down) {
        return 0;
    }
    // the blocks read ids, weights and scales that the output must not cover
    const int output_idx = i + match.node_count;
    if (ggml_cuda_tensors_overlap(match.dst, ids) || ggml_cuda_tensors_overlap(match.dst, match.weights) ||
            (match.expert_scale && ggml_cuda_tensors_overlap(match.dst, match.expert_scale)) ||
            !ggml_cuda_check_fusion_memory_ranges(cgraph, i + 1, match.node_count, &output_idx, 1)) {
        return 0;
    }
    ggml_cuda_mm_fusion_args_host fusion{};
    fusion.moe_out     = match.dst;
    fusion.moe_weights = match.weights;
    fusion.moe_scale   = match.expert_scale;
    ggml_cuda_mul_mat_vec_q(ctx, w, y, ids, down, &fusion);

    static const bool verify = getenv("GGML_CUDA_MOE_REDUCE_VERIFY") != nullptr;
    if (verify) {
        const size_t n = ggml_nelements(match.dst);
        std::vector<float> a(n), b(n);
        CUDA_CHECK(cudaMemcpyAsync(a.data(), match.dst->data, n*sizeof(float), cudaMemcpyDeviceToHost, ctx.stream()));
        ggml_cuda_mul_mat_vec_q(ctx, w, y, ids, down);
        ggml_cuda_op_moe_weighted_reduction(ctx, match.experts, match.expert_scale, match.weights, match.dst);
        CUDA_CHECK(cudaMemcpyAsync(b.data(), match.dst->data, n*sizeof(float), cudaMemcpyDeviceToHost, ctx.stream()));
        CUDA_CHECK(cudaStreamSynchronize(ctx.stream()));
        int nbad = 0;
        for (size_t k = 0; k < n; k++) {
            nbad += memcmp(&a[k], &b[k], sizeof(float)) != 0;
        }
        static int nprint = 0;
        if (nbad || nprint < 8) {
            nprint++;
            fprintf(stderr, "moe reduce verify: %s: %d of %zu differ\n", match.dst->name, nbad, n);
        }
    }
    return match.node_count;
}
